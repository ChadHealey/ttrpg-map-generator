import { partitionCoast, sampleCoast } from '../issue-172/coast-partition.mjs';
export function buildCoast(id, { anatomy = [0, 0], variation = 0 } = {}) {
  const anchors = [
    [0.68, -0.1],
    [0.49, -0.08],
    [0.35, 0.02],
    [0.45, 0.13],
    [0.7, 0.12],
    [0.74, 0.25],
    [0.6, 0.4],
    [0.47, 0.66],
    [0.18, 0.81],
    [-0.18, 0.82],
    [-0.48, 0.65],
    [-0.61, 0.4],
    [-0.64, 0.24],
    [-0.61, 0.065],
    [-0.78, 0.155],
    [-0.92, 0.09],
    [-0.96, 0.03],
    [-0.94, -0.14],
    [-0.75, -0.2],
    [-0.61, -0.065],
    [-0.62, -0.24],
    [-0.49, -0.43],
    [-0.37, -0.66],
    [-0.13, -0.83],
    [0.06, -0.79],
    [0.18, -0.49],
    [0.44, -0.42],
    [0.61, -0.3],
  ];
  const [u, v] = anatomy.map((x) => x * [1, 0.85, 0.6, 0.3][variation]);
  anchors[7] = [anchors[7][0] + 0.03 * u, anchors[7][1] + 0.015 * v];
  anchors[8] = [anchors[8][0] + 0.04 * v, anchors[8][1] + 0.035 * u];
  anchors[9] = [anchors[9][0] - 0.025 * u, anchors[9][1] - 0.025 * v];
  anchors[23] = [anchors[23][0] + 0.04 * v, anchors[23][1] + 0.025 * u];
  const coast = sampleCoast(anchors, { steps: 3, tension: 0.12 });
  const split = partitionCoast(id, {
    coast,
    roles: [
      { kind: 'lobe', start: 18, end: 33, far: [21, 30], disk: [0, 0.525] },
      { kind: 'lobe', start: 63, end: 75, far: [66, 72], disk: [-0.15, -0.56] },
      { kind: 'peninsula', start: 39, end: 57, far: [45, 51], disk: [-0.73, 0] },
    ],
    bay: { start: 0, end: 12, witness: [0.43, 0.02] },
    interiorWitness: [0, 0],
    islandAnchors: [60, 62, 76, 78, 79, 80],
  });
  split.bay.polygon.reverse();
  split.bay.mouthKind = 'wedge-geodesic';
  return split;
}
