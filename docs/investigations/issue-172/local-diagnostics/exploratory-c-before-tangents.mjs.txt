/** Direct northern shelf and southern taper; witnesses are fixed before quota fitting. */
const cubic = (a, b, c, d, steps = 4) =>
  Array.from({ length: steps }, (_, i) => {
    const t = (i + 1) / steps,
      u = 1 - t;
    return i === steps - 1
      ? d
      : [0, 1].map(
          (k) => u * u * u * a[k] + 3 * u * u * t * b[k] + 3 * u * t * t * c[k] + t * t * t * d[k],
        );
  });
export function buildCoast(id) {
  const la = [-0.44, 0.33],
    lb = [0.45, 0.3],
    lf0 = [-0.49, 0.48],
    lf1 = [0.46, 0.44];
  const sa = [-0.32, -0.4],
    sb = [0.23, -0.39],
    sf0 = [-0.36, -0.52],
    sf1 = [0.17, -0.55];
  const pa = [-0.5, -0.0525],
    pb = [-0.51, 0.0525],
    pf0 = [-0.66, -0.18],
    pf1 = [-0.66, 0.075];
  const inner = [0.32, 0],
    outer = [0.55, 0];
  const pocket = [
    inner,
    ...cubic(inner, [0.29, 0.1], [0.33, 0.205], [0.425, 0.22]),
    ...cubic([0.425, 0.22], [0.51, 0.235], [0.56, 0.1], outer),
  ];
  const northern = [
    la,
    ...cubic(la, [-0.48, 0.34], [-0.5, 0.42], lf0),
    ...cubic(lf0, [-0.48, 0.6], [-0.29, 0.66], [-0.15, 0.67]),
    ...cubic([-0.15, 0.67], [0.1, 0.71], [0.34, 0.65], [0.4, 0.55]),
    ...cubic([0.4, 0.55], [0.43, 0.52], [0.45, 0.48], lf1, 2),
    ...cubic(lf1, [0.48, 0.38], [0.46, 0.31], lb, 3),
  ];
  const southern = [
    sa,
    ...cubic(sa, [-0.37, -0.43], [-0.39, -0.48], sf0),
    ...cubic(sf0, [-0.28, -0.65], [-0.11, -0.71], [-0.03, -0.67]),
    ...cubic([-0.03, -0.67], [0.04, -0.64], [0.12, -0.59], sf1),
    ...cubic(sf1, [0.2, -0.49], [0.2, -0.43], sb, 3),
  ];
  const peninsula = [
    pa,
    ...cubic(pa, [-0.54, -0.09], [-0.59, -0.13], pf0),
    ...cubic(pf0, [-0.72, -0.155], [-0.79, -0.14], [-0.792, -0.065]),
    ...cubic([-0.792, -0.065], [-0.8, 0.025], [-0.76, 0.085], pf1),
    ...cubic(pf1, [-0.61, 0.075], [-0.56, 0.06], pb, 3),
  ];
  const interior = [
    outer,
    ...cubic(outer, [0.62, 0.05], [0.63, 0.1], [0.6, 0.15]),
    ...cubic([0.6, 0.15], [0.57, 0.18], [0.51, 0.23], lb, 3),
    la,
    ...cubic(la, [-0.5, 0.23], [-0.54, 0.13], pb),
    pa,
    ...cubic(pa, [-0.49, -0.15], [-0.5, -0.24], [-0.46, -0.29]),
    ...cubic([-0.46, -0.29], [-0.41, -0.34], [-0.37, -0.38], sa, 3),
    sb,
    ...cubic(sb, [0.33, -0.37], [0.41, -0.3], [0.46, -0.23]),
    ...cubic([0.46, -0.23], [0.48, -0.15], [0.32, -0.1], inner),
    ...pocket.slice(1, -1),
  ];
  return {
    interior,
    interiorWitness: [-0.02, -0.08],
    // Compact exposed lower-core margins, avoiding the far end of the southern lobe.
    islandAnchorEdges: [
      [interior[interior.indexOf(pa) + 1], interior[interior.indexOf(pa) + 2]],
      [interior[interior.indexOf(pa) + 3], interior[interior.indexOf(pa) + 4]],
      [interior[interior.indexOf(pa) + 5], interior[interior.indexOf(pa) + 6]],
      [southern[1], southern[2]],
      [interior[interior.indexOf(sb) + 1], interior[interior.indexOf(sb) + 2]],
      [interior[interior.indexOf(sb) + 3], interior[interior.indexOf(sb) + 4]],
    ],
    attachments: [
      {
        id: `${id}/lobe-1`,
        kind: 'lobe',
        root: [la, lb],
        polygon: northern,
        collar: { far: [lf0, lf1], disk: [0, 0.388] },
      },
      {
        id: `${id}/lobe-2`,
        kind: 'lobe',
        root: [sa, sb],
        polygon: southern,
        collar: { far: [sf0, sf1], disk: [-0.07, -0.46] },
      },
      {
        id: `${id}/peninsula`,
        kind: 'peninsula',
        root: [pa, pb],
        polygon: peninsula,
        collar: { far: [pf0, pf1], disk: [-0.595, -0.01] },
      },
    ],
    bay: { polygon: pocket, mouth: [inner, outer], witness: [0.425, 0.2] },
  };
}
